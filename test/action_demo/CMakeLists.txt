cmake_minimum_required(VERSION 3.16)
project(ros2_trace_viewer_action_demo LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tracetools REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LTTNG_UST REQUIRED IMPORTED_TARGET lttng-ust)
add_executable(action_demo demo.cpp action_provider.c)
target_compile_features(action_demo PRIVATE cxx_std_17)
target_include_directories(action_demo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
ament_target_dependencies(action_demo rclcpp rclcpp_action example_interfaces std_msgs tracetools)
target_link_libraries(action_demo PkgConfig::LTTNG_UST)
target_compile_options(action_demo PRIVATE -Wall -Wextra)
